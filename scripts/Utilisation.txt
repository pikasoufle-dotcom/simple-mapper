Installation initiale
bash
# 1. Installer la dépendance de build
npm install

# 2. (Optionnel) Pré-télécharger les vendor deps
npm run vendor
Build
bash
# Build les deux versions (offline + lite)
npm run build

# Build offline uniquement (recommandé pour événements)
npm run build:offline

# Build lite uniquement (CDN, plus petit)
npm run build:lite

# Build sans supprimer dist/ (conserve les anciens)
npm run build:keep

# Nettoyer dist/
npm run clean
Tester le résultat
bash
npm run serve:dist
# Ouvrir http://localhost:8000/simple-mapper-offline-v1.0.0/index.html
